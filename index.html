function parseTime(hhmm){ const [h,m]=hhmm.split(":").map(Number); return h*60+m; }

function fmtTime(hhmm){

  const mins=parseTime(hhmm); const h24=Math.floor(mins/60), m=mins%60;

function daysUntil(dateStr){

  const now=new Date(); now.setHours(0,0,0,0);

  const d=new Date(dateStr); d.setHours(0,0,0,0);

  return Math.round((d-now)/86400000);

}

function nextRenewalDate(sub){

  const cadenceDays = sub.cadence==="weekly"?7:sub.cadence==="yearly"?365:30;

  let d=new Date(sub.renewsOn); const now=new Date();

  while(d<now) d.setDate(d.getDate()+cadenceDays);

  return d;

}

function computeSemesterPace(semester, budgetTx){

  const start=new Date(semester.startDate), end=new Date(semester.endDate), now=new Date();

  const totalDays=Math.max(1,(end-start)/86400000);

  const elapsedDays=Math.min(totalDays, Math.max(0,(now-start)/86400000));

  const spent=budgetTx.filter(t=>new Date(t.date)>=start).reduce((s,t)=>s+t.amount,0);

  const remaining=semester.total-spent;

  const expectedRemaining=semester.total*(1-elapsedDays/totalDays);

  const weeklyAllowance=semester.total/(totalDays/7);

  let status="ok";

  if(remaining<expectedRemaining*0.75) status="over";

  else if(remaining<expectedRemaining*0.9) status="warn";

  return {remaining, expectedRemaining, weeklyAllowance, status, spent, totalDays, elapsedDays};

}

function startOfWeek(d){ const date=new Date(d); const day=date.getDay(); date.setDate(date.getDate()+(day===0?-6:1-day)); date.setHours(0,0,0,0); return date; }

function endOfWeek(d){ const s=startOfWeek(d); const e=new Date(s); e.setDate(e.getDate()+6); e.setHours(23,59,59,999); return e; }

const MAP_FT_W=2200, MAP_FT_H=1400, WALK_FT_PER_MIN=250;

function walkMinutes(pinA,pinB){

  if(!pinA||!pinB) return null;

  const dx=((pinA.x-pinB.x)/100)*MAP_FT_W, dy=((pinA.y-pinB.y)/100)*MAP_FT_H;

  return Math.max(1, Math.ceil(Math.sqrt(dx*dx+dy*dy)/WALK_FT_PER_MIN));

}

function runChecks(){

  const now=new Date(), todayStr=now.toDateString(), dayIdx=now.getDay();

  const nowMin=now.getHours()*60+now.getMinutes();

  const todays=state.classes.filter(c=>c.days.includes(dayIdx)).sort((a,b)=>parseTime(a.start)-parseTime(b.start));

  todays.forEach(c=>{

    const startMin=parseTime(c.start);

    if(nowMin>=startMin-15 && nowMin<startMin){

      pushAlert(`class-${c.id}-${todayStr}`, `${c.name} starts in ${startMin-nowMin} min${c.location?` at ${c.location}`:""}.`, "info");

    }

  });

  for(let i=0;i<todays.length-1;i++){

    const a=todays[i], b=todays[i+1];

    const gap=parseTime(b.start)-parseTime(a.end);

    const pinA=state.pins.find(p=>p.id===a.pinId), pinB=state.pins.find(p=>p.id===b.pinId);

    const wm=walkMinutes(pinA,pinB);

    if(wm!==null && gap<wm+5){

      pushAlert(`gap-${a.id}-${b.id}-${todayStr}`, `Tight transition: ${a.name} → ${b.name} is a ${wm} min walk but you only have ${gap} min.`, "warn");

    }

  }

  state.events.forEach(ev=>{

    if(!ev.date || !ev.time) return;

    if(new Date(ev.date).toDateString()!==todayStr) return;

    const startMin=parseTime(ev.time);

    if(nowMin>=startMin-60 && nowMin<startMin){

      pushAlert(`event-${ev.id}-${todayStr}`, `${ev.title} starts in ${startMin-nowMin} min${ev.location?` at ${ev.location}`:""}.`, "info");

    }

  });

  state.subscriptions.forEach(sub=>{

    const nd=nextRenewalDate(sub), away=daysUntil(nd);

    if(away>=0 && away<=3){

      pushAlert(`sub-${sub.id}-${nd.toDateString()}`, `${sub.name} renews ${away===0?"today":`in ${away} day${away>1?"s":""}`} — ${fmtMoney(sub.amount)}.`, "warn");

    }

  });

  state.importantDates.forEach(d=>{

    const away=daysUntil(d.date);

    if(away>=0 && away<=3){

      pushAlert(`date-${d.id}-${d.date}`, `${d.label} is ${away===0?"today":`in ${away} day${away>1?"s":""}`}.`, away<=1?"over":"warn");

    }

  });

  state.assignments.forEach(a=>{

    if(a.done) return;

    const away=daysUntil(a.dueDate);

    if(away>=0 && away<=2){

      pushAlert(`assign-${a.id}-${a.dueDate}`, `${a.title} is due ${away===0?"today":`in ${away} day${away>1?"s":""}`}.`, away===0?"over":"warn");

    }

  });

  render();

}


function buildBlueBonnetContext(){

  const now=new Date(), dayIdx=now.getDay();

  const todays=state.classes.filter(c=>c.days.includes(dayIdx)).sort((a,b)=>parseTime(a.start)-parseTime(b.start));

  const dueSoon=state.assignments.filter(a=>!a.done && daysUntil(a.dueDate)>=0 && daysUntil(a.dueDate)<=5).sort((a,b)=>new Date(a.dueDate)-new Date(b.dueDate));

  const todayEvents=state.events.filter(e=>e.date && new Date(e.date).toDateString()===now.toDateString());

  const discCats=state.budgetCats.filter(c=>c.discretionary);

  const discSpent=discCats.reduce((s,c)=>s+spentFor(c.id),0);

  const discLimit=discCats.reduce((s,c)=>s+c.weeklyLimit,0);

  const pace=state.semester?computeSemesterPace(state.semester, state.budgetTx):null;

  const upcomingSubs=state.subscriptions.map(s=>({...s, away:daysUntil(nextRenewalDate(s))})).filter(s=>s.away<=7);

  const lines=[];

  lines.push(`Today is ${now.toLocaleDateString(undefined,{weekday:"long",month:"long",day:"numeric"})}.`);

  lines.push(`Student is ${isVerified()?"a verified":"a pending, not-yet-verified"} student at ${state.enrollment?.school||"an unspecified school"}.`);

  lines.push(todays.length?`Today's classes: ${todays.map(c=>`${c.name} ${fmtTime(c.start)}-${fmtTime(c.end)}${c.location?` at ${c.location}`:""}`).join("; ")}.`:"No classes today.");

  lines.push(dueSoon.length?`Assignments due soon: ${dueSoon.map(a=>`${a.title} (due in ${daysUntil(a.dueDate)}d)`).join("; ")}.`:"No assignments due in the next 5 days.");

  lines.push(todayEvents.length?`Events today: ${todayEvents.map(e=>e.title).join(", ")}.`:"No events today.");

  lines.push(`Discretionary spending this week: $${discSpent.toFixed(2)} of $${discLimit.toFixed(2)} budgeted.`);

  if(pace) lines.push(`Semester funds: $${pace.remaining.toFixed(2)} left of $${state.semester.total}, about $${pace.weeklyAllowance.toFixed(2)}/week to stay on pace.`);

  if(upcomingSubs.length) lines.push(`Subscriptions renewing soon: ${upcomingSubs.map(s=>`${s.name} in ${s.away}d ($${s.amount})`).join("; ")}.`);

  return lines.join(" ");

}

async function sendBlueBonnet(){

  const inputEl=document.getElementById("bb-input");

  const text=(inputEl?.value||"").trim();

  if(!text || state.bbLoading) return;

  state.bbMessages=[...state.bbMessages, {role:"user", content:text}];

  state.bbLoading=true; state.bbError="";

  render();

  if(!BLUE_BONNET_ENDPOINT){

    state.bbMessages=[...state.bbMessages, {role:"assistant", content:"Blue Bonnet isn't wired up to a live endpoint yet. Set BLUE_BONNET_ENDPOINT near the top of this file's <script> to your actual Blue Bonnet API URL, and this chat will call it automatically."}];

    state.bbLoading=false; render(); return;

  }

  try{

    const res=await fetch(BLUE_BONNET_ENDPOINT, {

      method:"POST",

      headers:{"Content-Type":"application/json"},

      body: JSON.stringify({ messages: state.bbMessages, context: buildBlueBonnetContext() }),

    });

    const data=await res.json();

    state.bbMessages=[...state.bbMessages, {role:"assistant", content: data.reply || data.content || "Sorry, I didn't catch that."}];

  }catch(err){

    state.bbError="Blue Bonnet couldn't respond right now — try again in a moment.";

  }

  state.bbLoading=false;

  render();

}

window.__handleIdPhoto=function(e){

  const file=e.target.files[0];

  if(!file) return;

  const reader=new FileReader();

  reader.onload=function(){

    const img=new Image();

    img.onload=function(){

      const maxSize=320, scale=Math.min(1, maxSize/Math.max(img.width,img.height));

      const canvas=document.createElement("canvas");

      canvas.width=img.width*scale; canvas.height=img.height*scale;

      const ctx=canvas.getContext("2d");

      ctx.drawImage(img,0,0,canvas.width,canvas.height);

      state.draftProfile = state.draftProfile || {...state.profile};

      state.draftProfile.photo = canvas.toDataURL("image/jpeg", 0.85);

      render();

    };

    img.onerror=function(){ const el=document.getElementById("id-photo-error"); if(el) el.textContent="Couldn't process that photo — try a different one."; };

    img.src=reader.result;

  };

  reader.onerror=function(){ const el=document.getElementById("id-photo-error"); if(el) el.textContent="Couldn't read that file."; };

  reader.readAsDataURL(file);

};

  if(!t) return;

  const action=t.dataset.action;


  switch(action){

    case "set-tab": state.tab=t.dataset.tab; render(); break;

    case "dismiss-banner": state.dismissed=[...state.dismissed, t.dataset.id]; render(); break;

    case "open-feed": state.modal={type:"feed"}; render(); break;

    case "delete-date": state.importantDates=state.importantDates.filter(d=>d.id!==t.dataset.id); persist("campus-dates", state.importantDates); render(); break;

    }

    case "add-checklist-item": window.__addChecklistItem(t.dataset.key); break;


    case "open-modal": {

      const type=t.dataset.modal;

      if(type==="add-class") state.draftClass={days:[], color:CLASS_COLORS[0]};

      if(type==="add-split") state.draftSplitOwe=false;

      state.modal={type, catId:t.dataset.catid};

      render(); break;

    }

    case "toggle-class-day": {

      const d=Number(t.dataset.day);

      const draft=state.draftClass||{days:[],color:CLASS_COLORS[0]};

      draft.days = draft.days.includes(d)? draft.days.filter(x=>x!==d) : [...draft.days, d];

      state.draftClass=draft; render(); break;

    }

    case "set-class-color": {

      const draft=state.draftClass||{days:[],color:CLASS_COLORS[0]};

      draft.color=t.dataset.color; state.draftClass=draft; render(); break;

    }

    case "submit-add-class": {

      const name=document.getElementById("cl-name").value.trim();

      const start=document.getElementById("cl-start").value;

      const end=document.getElementById("cl-end").value;

      const pinId=document.getElementById("cl-pin").value;

      const draft=state.draftClass||{days:[],color:CLASS_COLORS[0]};

      if(!name || draft.days.length===0) return;

      const pin=state.pins.find(p=>p.id===pinId);

      state.classes=[...state.classes, {id:uid(), name, days:draft.days, start, end, pinId:pinId||null, location:pin?pin.name:"", color:draft.color}];

      persist("campus-classes", state.classes);

      state.modal=null; state.draftClass=null; render(); break;

    }

    case "submit-add-date": {

      const label=document.getElementById("md-label").value.trim();

      const date=document.getElementById("md-date").value;

      if(!label) return;

      state.importantDates=[...state.importantDates, {id:uid(), label, date}];

      persist("campus-dates", state.importantDates);

      state.modal=null; render(); break;

    }

    case "submit-add-assignment": {

      const title=document.getElementById("ma-title").value.trim();

      const dueDate=document.getElementById("ma-due").value;

      if(!title) return;

      state.assignments=[...state.assignments, {id:uid(), title, dueDate, done:false}];

      persist("campus-assignments", state.assignments);

      state.modal=null; render(); break;

    }


    case "toggle-budget-cat": state.expandedBudgetCat = state.expandedBudgetCat===t.dataset.id?null:t.dataset.id; render(); break;

    case "delete-category": {

      state.budgetCats=state.budgetCats.filter(c=>c.id!==t.dataset.id);

      persist("campus-budget-cats", state.budgetCats); render(); break;

    }

    case "submit-add-category": {

      const name=document.getElementById("mc-name").value.trim();

      const limit=parseFloat(document.getElementById("mc-limit").value);

      const disc=document.getElementById("mc-disc").checked;

      if(!name || !limit || limit<=0) return;

      state.budgetCats=[...state.budgetCats, {id:uid(), name, weeklyLimit:limit, discretionary:disc}];

      persist("campus-budget-cats", state.budgetCats);

      state.modal=null; render(); break;

    }

    case "delete-tx": state.budgetTx=state.budgetTx.filter(t2=>t2.id!==t.dataset.id); persist("campus-budget-tx", state.budgetTx); render(); break;

    case "submit-add-tx": {

      const catId=t.dataset.catid;

      const amount=parseFloat(document.getElementById("mt-amount").value);

      const note=document.getElementById("mt-note").value;

      const cat=state.budgetCats.find(c=>c.id===catId);

      if(!cat || !amount || amount<=0) return;

      const ws=startOfWeek(new Date()), we=endOfWeek(new Date());

      const prevSpent=state.budgetTx.filter(tx=>tx.categoryId===catId).filter(tx=>{const d=new Date(tx.date); return d>=ws&&d<=we;}).reduce((s,tx)=>s+tx.amount,0);

      state.budgetTx=[...state.budgetTx, {id:uid(), categoryId:catId, amount, note:note||"", date:new Date().toISOString()}];

      persist("campus-budget-tx", state.budgetTx);

      const newSpent=prevSpent+amount, prevPct=(prevSpent/cat.weeklyLimit)*100, newPct=(newSpent/cat.weeklyLimit)*100;

      if(prevPct<100 && newPct>=100) pushAlert(`budget-over-${cat.id}-${Date.now()}`, `${cat.name} hit its weekly limit of ${fmtMoney(cat.weeklyLimit)}.`, "over");

      else if(prevPct<75 && newPct>=75) pushAlert(`budget-warn-${cat.id}-${Date.now()}`, `${cat.name} is at ${Math.round(newPct)}% of its weekly limit.`, "warn");

      state.modal=null; render(); break;

    }

    case "submit-semester": {

      const total=parseFloat(document.getElementById("ss-total").value);

      const startDate=document.getElementById("ss-start").value;

      const endDate=document.getElementById("ss-end").value;

      if(!total || total<=0 || !endDate) return;

      state.semester={total, startDate, endDate};

      persist("campus-semester", state.semester);

      state.modal=null; render(); break;

    }

    case "clear-semester": state.semester=null; persist("campus-semester", null); state.modal=null; render(); break;

    case "delete-sub": state.subscriptions=state.subscriptions.filter(s=>s.id!==t.dataset.id); persist("campus-subscriptions", state.subscriptions); render(); break;

    case "submit-add-sub": {

      const name=document.getElementById("ms-name").value.trim();

      const amount=parseFloat(document.getElementById("ms-amount").value);

      const cadence=document.getElementById("ms-cadence").value;

      const renewsOn=document.getElementById("ms-renews").value;

      if(!name || !amount || amount<=0) return;

      state.subscriptions=[...state.subscriptions, {id:uid(), name, amount, cadence, renewsOn}];

      persist("campus-subscriptions", state.subscriptions);

      state.modal=null; render(); break;

    }

    case "settle-split": {

      const person=t.dataset.person;

      state.splits=state.splits.map(s=>s.person===person?{...s,settled:true}:s);

      persist("campus-splits", state.splits); render(); break;

    }

    case "set-split-direction": {

      state.draftSplitOwe = t.dataset.owe==="true";

      const theyBtn=document.getElementById("msp-btn-they"), iBtn=document.getElementById("msp-btn-i");

      if(theyBtn && iBtn){

        theyBtn.style.background = state.draftSplitOwe? "white":"#3B4B6B"; theyBtn.style.color = state.draftSplitOwe? "#1B2430":"white";

        iBtn.style.background = state.draftSplitOwe? "#3B4B6B":"white"; iBtn.style.color = state.draftSplitOwe? "white":"#1B2430";

      }

      break;

    }

    case "submit-add-split": {

      const description=document.getElementById("msp-desc").value.trim();

      const amount=parseFloat(document.getElementById("msp-amount").value);

      const person=document.getElementById("msp-person").value.trim();

      if(!description || !amount || amount<=0 || !person) return;

      state.splits=[...state.splits, {id:uid(), description, amount, person, iOwe:!!state.draftSplitOwe, settled:false, date:new Date().toISOString()}];

      persist("campus-splits", state.splits);

      state.modal=null; render(); break;

    }

    case "delete-event": state.events=state.events.filter(ev=>ev.id!==t.dataset.id); persist("campus-events", state.events); render(); break;

    case "submit-add-event": {

      const title=document.getElementById("me-title").value.trim();

      const date=document.getElementById("me-date").value;

      const time=document.getElementById("me-time").value;

      const pinId=document.getElementById("me-pin").value;

      const note=document.getElementById("me-note").value;

      if(!title) return;

      const pin=state.pins.find(p=>p.id===pinId);

      state.events=[...state.events, {id:uid(), title, date, time, pinId:pinId||null, location:pin?pin.name:"", note}];

      persist("campus-events", state.events);

      state.modal=null; render(); break;

    }


    case "toggle-map-add": state.mapAddMode=!state.mapAddMode; render(); break;

    case "delete-pin": state.pins=state.pins.filter(p=>p.id!==t.dataset.id); persist("campus-pins", state.pins); render(); break;

    case "submit-name-pin": {

      const name=document.getElementById("mp-name").value.trim();

      if(!name || !state.pendingPin) return;

      state.pins=[...state.pins, {id:uid(), name, x:state.pendingPin.x, y:state.pendingPin.y}];

      persist("campus-pins", state.pins);

      state.pendingPin=null; state.modal=null; render(); break;

    }


    case "trigger-photo": document.getElementById("id-photo-file").click(); break;

    case "save-id": {

      const dp=state.draftProfile||state.profile;

      state.profile={ ...state.profile, name:dp.name||"", school:dp.school||"", studentId:dp.studentId||"", photo:dp.photo||state.profile.photo||"" };

      persist("campus-profile", state.profile);

      state.draftProfile=null; state.modal=null; render(); break;

    }

    case "sign-out": state.enrollment=null; persist("campus-enrollment", null); state.modal=null; render(); break;


    case "send-bluebonnet": sendBlueBonnet(); break;

  }

});


document.addEventListener("click", function(e){

  const svg=document.getElementById("campus-map-svg");

  if(!svg || !state.mapAddMode) return;

  if(e.target!==svg && e.target.closest("svg")!==svg) return;

  if(e.target.closest("[data-action]")) return;

  const rect=svg.getBoundingClientRect();

  const x=((e.clientX-rect.left)/rect.width)*100;

  const y=((e.clientY-rect.top)/rect.height)*100;

  state.pendingPin={x,y};

  state.mapAddMode=false;

  state.modal={type:"name-pin"};

  render();

});
